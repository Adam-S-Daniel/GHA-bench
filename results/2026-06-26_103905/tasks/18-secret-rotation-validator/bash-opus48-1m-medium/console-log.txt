#!/usr/bin/env bash
#
# secret-rotation-validator.sh
#
# Reads a JSON configuration of secrets (name, last_rotated date,
# rotation_policy_days, required_by services) and classifies each secret by
# rotation urgency:
#   - expired : the rotation deadline is already in the past
#   - warning : the rotation deadline falls within the warning window
#   - ok      : the rotation deadline is comfortably in the future
#
# It then emits a rotation report grouped by urgency in either a markdown
# table or JSON.
#
set -euo pipefail

# Classify a single secret given days_until_expiry and the warning window.
# Echoes one of: expired | warning | ok
classify() {
  local days_until="$1" warning="$2"
  if (( days_until < 0 )); then
    echo "expired"
  elif (( days_until <= warning )); then
    echo "warning"
  else
    echo "ok"
  fi
}

main() {
  local config="" today="" warning_days=14 format="markdown"

  while (( $# > 0 )); do
    case "$1" in
      --config)       config="$2"; shift 2 ;;
      --today)        today="$2"; shift 2 ;;
      --warning-days) warning_days="$2"; shift 2 ;;
      --format)       format="$2"; shift 2 ;;
      *) echo "error: unknown argument: $1" >&2; return 2 ;;
    esac
  done

  # Default "today" to the system date if not supplied (testability hook).
  [[ -z "$today" ]] && today="$(date +%F)"

  local today_epoch
  today_epoch="$(date -d "$today" +%s)"

  # Build an enriched JSON array via jq, computing the per-secret status.
  jq --argjson today "$today_epoch" --argjson warn "$warning_days" '
    def daysuntil(lr; pol):
      ((lr | strptime("%Y-%m-%d") | mktime) + (pol * 86400) - $today) / 86400 | floor;
    {
      secrets: [
        .[] | . as $s
        | (daysuntil(.last_rotated; .rotation_policy_days)) as $du
        | {
            name: .name,
            last_rotated: .last_rotated,
            rotation_policy_days: .rotation_policy_days,
            required_by: .required_by,
            days_until_expiry: $du,
            status: (if $du < 0 then "expired"
                     elif $du <= $warn then "warning"
                     else "ok" end)
          }
      ]
    }
  ' "$config"
}

main "$@"
