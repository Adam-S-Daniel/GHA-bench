@dataclass(frozen=True)
class FlakyTest:
    """A test that both passed and failed across the aggregated runs."""
    full_name: str
    passes: int
    failures: int


@dataclass(frozen=True)
class Summary:
    total: int
    passed: int
    failed: int
    skipped: int
    duration: float
    flaky: list[FlakyTest]


def aggregate(runs: list[list[TestResult]]) -> Summary:
    """Aggregate result lists from multiple runs (matrix jobs) into a Summary.

    Totals count every executed test case (the same test in N runs counts N
    times, matching what actually ran in CI). Flaky detection correlates by
    full_name: a test is flaky iff it has >=1 pass and >=1 failure.
    """
    counts = {"passed": 0, "failed": 0, "skipped": 0}
    duration = 0.0
    outcomes: dict[str, dict[str, int]] = {}
    for run in runs:
        for r in run:
            counts[r.status] += 1
            duration += r.duration
            per_test = outcomes.setdefault(r.full_name, {"passed": 0, "failed": 0})
            if r.status in per_test:
                per_test[r.status] += 1

    flaky = [
        FlakyTest(name, o["passed"], o["failed"])
        for name, o in sorted(outcomes.items())
        if o["passed"] and o["failed"]
    ]
    return Summary(
        total=sum(counts.values()),
        passed=counts["passed"],
        failed=counts["failed"],
        skipped=counts["skipped"],
        duration=duration,
        flaky=flaky,
    )
EOF

def collect_result_files(target: str | Path) -> list[Path]:
    """Resolve a directory (or single file) into a sorted list of result files."""
    target = Path(target)
    if target.is_dir():
        files = sorted(p for p in target.iterdir() if p.suffix in (".xml", ".json"))
        if not files:
            raise AggregatorError(
                f"{target}: no result files (*.xml, *.json) found in directory")
        return files
    return [target]


def main(argv: list[str] | None = None) -> int:
    """CLI: aggregate result files and emit the markdown summary.

    Prints machine-readable TOTALS/FLAKY lines to stdout for CI assertions,
    writes markdown to --output, and appends it to $GITHUB_STEP_SUMMARY when
    running under GitHub Actions.
    """

    import os
    import sys

        description="Aggregate JUnit XML / JSON test results into a markdown summary.")

                        help="result files or directories containing them")

**What was built (Python):**
- `aggregator.py` — parses JUnit XML (`<testsuite>`/`<testsuites>`, failure/error/skipped) and a JSON results format into a uniform `TestResult` shape, aggregates across files as matrix runs, computes totals (passed/failed/skipped/duration), detects flaky tests (≥1 pass and ≥1 fail for the same test across runs), and renders a markdown job summary. The CLI prints machine-readable `TOTALS`/`FLAKY` lines, writes `--output summary.md`, and appends the markdown to `$GITHUB_STEP_SUMMARY`. All bad inputs (missing file, invalid XML/JSON, unknown status/extension, empty directory) raise `AggregatorError` with the file path in the message and exit 2.
- `fixtures/` — three sample matrix-job files (2 JUnit XML + 1 JSON) containing one flaky test, two failures, and a skip.
- `tests/` — 26 tests built in five strict red/green TDD cycles, each with a demonstrated failing run before implementation and one git commit per cycle (see `git log`). Includes workflow structure tests: YAML triggers/permissions/jobs/`needs`, referenced paths exist, and actionlint exits 0.
- `.github/workflows/test-results-aggregator.yml` — push/PR/schedule/dispatch triggers, `contents: read` permissions, a `unit-tests` job (checkout, pip install, pytest) and a dependent `aggregate-results` job that runs the aggregator over the fixtures and publishes the job summary. `RESULTS_DIR` (default `fixtures`) lets the harness inject per-case data.
