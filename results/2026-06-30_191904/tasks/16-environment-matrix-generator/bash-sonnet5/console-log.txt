  | ($entries | map(.key)) as $keys
  | ($entries | map(.value)) as $vals
  | ($vals | combinations) as $combo
  | [$keys, $combo] | transpose | map({(.[0]): .[1]}) | add;

# combo matches rule if every key in rule that also exists in combo has an equal value
def matches_rule(combo; rule):
  (rule | to_entries) as $kvs
  | ($kvs | all( .value == (combo[.key]) ));

def apply_exclude(combos; excludes):
  combos | map( . as $c | select( ( [excludes[] | matches_rule($c; .)] | any ) | not ) );

# GH include semantics: for each include rule (in order), find combos where merging
# would not overwrite an *original axis* value (i.e. no conflicting existing key).
# If merge is compatible with >=1 combo, merge (allowing overwrite of previously
# include-added keys) into each; else add as a standalone new combo.
def apply_include(combos; axisKeys; includes):
  reduce includes[] as $inc (
    combos;
    . as $cur
    | ( [ $cur[] | select( matches_rule(.; $inc) ) ] ) as $matched
    | if ( $matched | length ) > 0 then
        ( $cur | map( if matches_rule(.; $inc) then . + $inc else . end ) )
      else
        $cur + [ $inc ]
      end
  );
EOF
